## Removing non-confirmed rodent sightings based off “descriptor” column and create zip index:

* ### Remove rows “signs of rodents” value in the “descriptor” column

* ### We believe signs of rodents do not equate to confirmed rodent presence

| CREATE OR REPLACE TABLE rodent\_confirmed\_sightings\_zip AS SELECT   incident\_zip                                                          AS zip\_code,   INITCAP(MAX(borough))                                                 AS borough,   COUNT(\*)                                                              AS rodent\_complaint\_count,   COUNT\_IF(descriptor \= 'Rat Sighting')                                 AS rat\_sighting\_count,   COUNT\_IF(descriptor \= 'Mouse Sighting')                               AS mouse\_sighting\_count,   COUNT\_IF(descriptor \= 'Signs of Rodents')                             AS signs\_of\_rodents\_count,   COUNT\_IF(status \= 'In Progress')                                      AS open\_complaint\_count,   COUNT\_IF(status \= 'Closed')                                           AS closed\_complaint\_count,   MIN(created\_date)                                                     AS earliest\_complaint\_date,   MAX(created\_date)                                                     AS latest\_complaint\_date,   COUNT\_IF(created\_date \>= DATE\_SUB(CURRENT\_DATE(), 90))                AS complaints\_last\_90d,   AVG(CASE WHEN closed\_date \<\> 'null'            THEN DATEDIFF(TO\_TIMESTAMP(closed\_date), created\_date) END)  AS avg\_days\_to\_close,   AVG(CAST(latitude  AS DOUBLE))                                        AS centroid\_lat,   AVG(CAST(longitude AS DOUBLE))                                        AS centroid\_lon FROM rodent\_sightings\_confirmed GROUP BY incident\_zip;  |

# Restaurant inspection data modification

## Turned “restaurant inspections” raw data to zip-indexed database: 

| CREATE OR REPLACE TABLE nyc\_data.default.restaurant\_zip\_metrics\_v4 AS WITH events AS (   SELECT     camis,     MAX(dba)                                                 AS dba,     CAST(inspection\_date AS DATE)                            AS inspection\_date,     CASE WHEN boro IN ('Manhattan','Brooklyn','Queens','Bronx','Staten Island')          THEN boro END                                       AS borough,     CASE WHEN zipcode RLIKE '^\[0-9\]{5}$' THEN zipcode END    AS zip\_normalized,     MAX(cuisine\_description)                                 AS cuisine\_description,     COUNT(violation\_code)                                    AS violation\_row\_count,     MAX(CAST(score AS INT))                                  AS score,     COUNT(DISTINCT score)                                    AS distinct\_score\_values,     MAX(grade)                                               AS grade,     MAX(violation\_code IN ('04K','04L'))                     AS has\_rodent\_violation,     MAX(violation\_code \= '04K')                              AS has\_rat\_violation,     MAX(violation\_code \= '04L')                              AS has\_mouse\_violation,     MAX(violation\_code \= '08A')                              AS has\_conducive\_condition,     MAX(critical\_flag \= 'Critical')                          AS has\_critical\_violation   FROM nyc\_data.default.restaurant\_inspections\_raw   GROUP BY camis, CAST(inspection\_date AS DATE), boro, zipcode ), e AS (   SELECT \* FROM events WHERE zip\_normalized IS NOT NULL ), zip\_metrics AS (   SELECT     zip\_normalized                                                     AS zip\_code,     MAX\_BY(borough, 1\)                                                 AS most\_common\_borough,     COUNT(DISTINCT camis)                                              AS distinct\_restaurant\_count,     COUNT(\*)                                                           AS inspection\_event\_count,     SUM(violation\_row\_count)                                           AS total\_violation\_rows,     CASE WHEN COUNT(DISTINCT camis) \< 5  THEN 'Very limited'          WHEN COUNT(DISTINCT camis) \< 10 THEN 'Limited'          ELSE 'Standard' END                                           AS sample\_size\_category,     COUNT(DISTINCT camis) \< 10                                         AS small\_sample\_warning,     COUNT\_IF(score IS NOT NULL)                                        AS scored\_event\_count,     COUNT\_IF(score IS NULL)                                            AS no\_score\_event\_count,     COUNT\_IF(distinct\_score\_values \> 1\)                                AS ambiguous\_score\_event\_count,     ROUND(AVG(score), 2\)                                               AS avg\_inspection\_score,     CAST(PERCENTILE\_CONT(0.5) WITHIN GROUP (ORDER BY score) AS DOUBLE) AS median\_inspection\_score,     MIN(score)                                                         AS min\_inspection\_score,     MAX(score)                                                         AS max\_inspection\_score,     COUNT\_IF(has\_rodent\_violation)                                     AS rodent\_evidence\_event\_count,     ROUND(COUNT\_IF(has\_rodent\_violation) \* 100.0 / COUNT(\*), 2\)        AS rodent\_evidence\_event\_pct,     COUNT(DISTINCT IF(has\_rodent\_violation, camis, NULL))              AS rodent\_evidence\_restaurant\_count,     ROUND(COUNT(DISTINCT IF(has\_rodent\_violation, camis, NULL)) \* 100.0           / COUNT(DISTINCT camis), 2\)                                  AS rodent\_evidence\_restaurant\_pct,     COUNT\_IF(has\_conducive\_condition)                                  AS conducive\_condition\_event\_count,     ROUND(COUNT\_IF(has\_conducive\_condition) \* 100.0 / COUNT(\*), 2\)     AS conducive\_condition\_event\_pct,     COUNT(DISTINCT IF(has\_conducive\_condition, camis, NULL))           AS conducive\_condition\_restaurant\_count,     ROUND(COUNT(DISTINCT IF(has\_conducive\_condition, camis, NULL)) \* 100.0           / COUNT(DISTINCT camis), 2\)                                  AS conducive\_condition\_restaurant\_pct,     COUNT\_IF(grade IN ('A','B','C'))                                   AS final\_graded\_event\_count,     COUNT\_IF(grade \= 'A')                                              AS a\_grade\_event\_count,     ROUND(COUNT\_IF(grade \= 'A') \* 100.0           / NULLIF(COUNT\_IF(grade IN ('A','B','C')), 0), 2\)            AS a\_grade\_event\_pct,     COUNT\_IF(has\_critical\_violation)                                   AS critical\_violation\_event\_count,     ROUND(COUNT\_IF(has\_critical\_violation) \* 100.0 / COUNT(\*), 2\)      AS critical\_violation\_event\_pct,     MIN(inspection\_date)                                               AS earliest\_inspection\_date,     MAX(inspection\_date)                                               AS latest\_inspection\_date   FROM e   GROUP BY zip\_normalized ), event\_followup AS (   SELECT     camis,     inspection\_date,     zip\_normalized,     has\_rodent\_violation,     LEAD(inspection\_date)       OVER (PARTITION BY camis ORDER BY inspection\_date) AS next\_inspection\_date,     LEAD(has\_rodent\_violation)  OVER (PARTITION BY camis ORDER BY inspection\_date) AS next\_has\_rodent   FROM e ), rodent\_hits AS (   SELECT     camis,     zip\_normalized,     DATEDIFF(next\_inspection\_date, inspection\_date) AS days\_to\_next,     next\_has\_rodent   FROM rodent\_hit\_events,     ROUND(       COUNT\_IF(days\_to\_next IS NOT NULL AND days\_to\_next \<= 90\) \* 100.0       / NULLIF(COUNT(\*), 0), 2\)                                       AS followup\_90d\_rate,     CAST(PERCENTILE\_CONT(0.5) WITHIN GROUP (ORDER BY days\_to\_next) AS INT)                                                                       AS median\_followup\_days,     ROUND(       COUNT(DISTINCT CASE WHEN next\_has\_rodent \= true THEN camis END) \* 100.0       / NULLIF(COUNT(DISTINCT CASE WHEN days\_to\_next IS NOT NULL THEN camis END), 0), 2\)                                                                       AS recurrence\_rate   FROM rodent\_hits   GROUP BY zip\_normalized ) SELECT   m.zip\_code,   m.most\_common\_borough,   m.distinct\_restaurant\_count, m.ambiguous\_score\_event\_count,   m.avg\_inspection\_score,   m.median\_inspection\_score,   m.min\_inspection\_score,   m.max\_inspection\_score,   m.rodent\_evidence\_event\_count,   m.rodent\_evidence\_event\_pct,   m.rodent\_evidence\_restaurant\_count,   m.rodent\_evidence\_restaurant\_pct,   m.conducive\_condition\_event\_count,   m.conducive\_condition\_event\_pct,   m.conducive\_condition\_restaurant\_count,   m.conducive\_condition\_restaurant\_pct,   m.final\_graded\_event\_count,   m.a\_grade\_event\_count,   m.a\_grade\_event\_pct, f.median\_followup\_days,   f.recurrence\_rate FROM zip\_metrics m LEFT JOIN zip\_followup f   ON m.zip\_code \= f.zip\_code;  |

# Combining the two datasets:

## Initial merge based off zip index:

| CREATE OR REPLACE TABLE nyc\_data.default.service\_gap\_index\_zip AS WITH base AS (   SELECT     i.zip\_code,     i.most\_common\_borough                                          AS borough,     r.centroid\_lat, r.centroid\_lon,     COALESCE(r.rodent\_complaint\_count, 0\)                          AS demand\_raw,     i.inspection\_event\_count / i.distinct\_restaurant\_count         AS coverage\_raw,     i.median\_followup\_days                                         AS followup\_days\_raw,     i.recurrence\_rate                                              AS recurrence\_raw,     i.rodent\_hit\_events,     i.distinct\_restaurant\_count,     i.small\_sample\_warning   FROM nyc\_data.default.restaurant\_zip\_metrics\_v4 i   LEFT JOIN nyc\_data.default.rodent\_sightings\_confirmed\_zip r USING (zip\_code) ), ranked AS (   SELECT \*,     PERCENT\_RANK() OVER (ORDER BY demand\_raw)                       AS demand\_pct,     PERCENT\_RANK() OVER (ORDER BY coverage\_raw)                     AS coverage\_pct,     1 \- PERCENT\_RANK() OVER (ORDER BY followup\_days\_raw)            AS followup\_pct,     1 \- PERCENT\_RANK() OVER (ORDER BY recurrence\_raw)               AS recurrence\_pct   FROM base ), scored AS (   SELECT \*,     (COALESCE(coverage\_pct, 0\) \+ COALESCE(followup\_pct, 0\) \+ COALESCE(recurrence\_pct, 0))       / (IF(coverage\_pct IS NULL,0,1) \+ IF(followup\_pct IS NULL,0,1) \+ IF(recurrence\_pct IS NULL,0,1))                                                                     AS response\_pct,     IF(followup\_pct IS NULL, 1, 0\) \+ IF(recurrence\_pct IS NULL, 1, 0\)                                                                     AS missing\_response\_components   FROM ranked ) SELECT   zip\_code, borough, centroid\_lat, centroid\_lon,   demand\_raw, coverage\_raw, followup\_days\_raw, recurrence\_raw,   ROUND(demand\_pct, 3\)                                              AS demand\_pct,   ROUND(coverage\_pct, 3\)                                            AS coverage\_pct,   ROUND(followup\_pct, 3\)                                            AS followup\_pct,   ROUND(recurrence\_pct, 3\)                                          AS recurrence\_pct,   ROUND(response\_pct, 3\)                                            AS response\_pct,   ROUND(demand\_pct \- response\_pct, 3\)                               AS service\_gap\_index,   RANK() OVER (ORDER BY demand\_pct \- response\_pct DESC)             AS gap\_rank,   CASE     WHEN rodent\_hit\_events IS NULL OR rodent\_hit\_events \< 10       OR distinct\_restaurant\_count \< 20 OR small\_sample\_warning THEN 'low'     WHEN missing\_response\_components \> 0 THEN 'partial'     ELSE 'ok'   END                                                               AS confidence FROM scored;  |

## Adding seasonal data:

* ### Multiplied out data from yearly granularity to seasonal granularity

| CREATE OR REPLACE TABLE nyc\_data.default.service\_gap\_index\_zip\_seasonal AS WITH insp\_events AS (   SELECT camis, zip\_normalized AS zip\_code, inspection\_date, has\_direct\_rodent\_evidence,     CASE WHEN MONTH(inspection\_date) IN (12,1,2) THEN 'winter'          WHEN MONTH(inspection\_date) IN (3,4,5)  THEN 'spring'          WHEN MONTH(inspection\_date) IN (6,7,8)  THEN 'summer'          ELSE 'fall' END AS season,     LEAD(inspection\_date)             OVER (PARTITION BY camis ORDER BY inspection\_date) AS next\_date,     LEAD(has\_direct\_rodent\_evidence)  OVER (PARTITION BY camis ORDER BY inspection\_date) AS next\_has\_rodent   FROM nyc\_data.default.restaurant\_inspection\_events\_v2   WHERE zip\_normalized IS NOT NULL ), insp\_season AS (   SELECT e.zip\_code, e.season,     COUNT(\*) / MAX(m.distinct\_restaurant\_count)                                    AS coverage\_raw,     COUNT\_IF(e.has\_direct\_rodent\_evidence)                                         AS rodent\_hit\_events,     CAST(PERCENTILE\_CONT(0.5) WITHIN GROUP (ORDER BY       CASE WHEN e.has\_direct\_rodent\_evidence THEN DATEDIFF(e.next\_date, e.inspection\_date) END) AS INT)                                                                                    AS followup\_days\_raw,     ROUND(COUNT(DISTINCT CASE WHEN e.has\_direct\_rodent\_evidence AND e.next\_has\_rodent THEN e.camis END) \* 100.0       / NULLIF(COUNT(DISTINCT CASE WHEN e.has\_direct\_rodent\_evidence AND e.next\_date IS NOT NULL THEN e.camis END), 0), 2\)                                                                                    AS recurrence\_raw   FROM insp\_events e   JOIN nyc\_data.default.restaurant\_zip\_metrics\_v4 m ON e.zip\_code \= m.zip\_code   GROUP BY e.zip\_code, e.season ), demand\_season AS (   SELECT incident\_zip AS zip\_code,     CASE WHEN MONTH(created\_date) IN (12,1,2) THEN 'winter'          WHEN MONTH(created\_date) IN (3,4,5)  THEN 'spring'          WHEN MONTH(created\_date) IN (6,7,8)  THEN 'summer'          ELSE 'fall' END AS season,     COUNT(\*) AS demand\_raw   FROM nyc\_data.default.rat\_sightings\_2   WHERE descriptor IN ('Rat Sighting', 'Mouse Sighting', 'Signs of Rodents')   GROUP BY incident\_zip, 2 ), base AS (   SELECT     s.season,     m.zip\_code,     m.most\_common\_borough                         AS borough,     r.centroid\_lat, r.centroid\_lon,     COALESCE(d.demand\_raw, 0\)                     AS demand\_raw,     i.coverage\_raw,     i.followup\_days\_raw,     i.recurrence\_raw,     i.rodent\_hit\_events,     m.distinct\_restaurant\_count,     m.small\_sample\_warning   FROM nyc\_data.default.restaurant\_zip\_metrics\_v4 m   CROSS JOIN (SELECT explode(array('winter','spring','summer','fall')) AS season) s   LEFT JOIN insp\_season   i ON i.zip\_code \= m.zip\_code AND i.season \= s.season   LEFT JOIN demand\_season d ON d.zip\_code \= m.zip\_code AND d.season \= s.season   LEFT JOIN nyc\_data.default.rodent\_sightings\_confirmed\_zip r ON r.zip\_code \= m.zip\_code ), ranked AS (   SELECT \*,     PERCENT\_RANK() OVER (PARTITION BY season ORDER BY demand\_raw)             AS demand\_pct,     PERCENT\_RANK() OVER (PARTITION BY season ORDER BY coverage\_raw)           AS coverage\_pct,     1 \- PERCENT\_RANK() OVER (PARTITION BY season ORDER BY followup\_days\_raw)  AS followup\_pct,     1 \- PERCENT\_RANK() OVER (PARTITION BY season ORDER BY recurrence\_raw)     AS recurrence\_pct   FROM base ), scored AS (   SELECT \*,     (COALESCE(coverage\_pct,0) \+ COALESCE(followup\_pct,0) \+ COALESCE(recurrence\_pct,0))       / NULLIF(IF(coverage\_pct IS NULL,0,1) \+ IF(followup\_pct IS NULL,0,1) \+ IF(recurrence\_pct IS NULL,0,1), 0\)                                                                               AS response\_pct   FROM ranked ), seasonal AS (   SELECT     season, zip\_code, borough, centroid\_lat, centroid\_lon,     demand\_raw, coverage\_raw, followup\_days\_raw, recurrence\_raw,     ROUND(demand\_pct,3) AS demand\_pct, ROUND(coverage\_pct,3) AS coverage\_pct,     ROUND(followup\_pct,3) AS followup\_pct, ROUND(recurrence\_pct,3) AS recurrence\_pct,     ROUND(response\_pct,3) AS response\_pct,     ROUND(demand\_pct \- response\_pct,3) AS service\_gap\_index,     RANK() OVER (PARTITION BY season ORDER BY demand\_pct \- response\_pct DESC) AS gap\_rank,     CASE       WHEN rodent\_hit\_events IS NULL OR rodent\_hit\_events \< 5         OR distinct\_restaurant\_count \< 20 OR small\_sample\_warning THEN 'low'       ELSE 'ok'     END AS confidence   FROM scored ) SELECT 'full' AS season, zip\_code, borough, centroid\_lat, centroid\_lon,        demand\_raw, coverage\_raw, followup\_days\_raw, recurrence\_raw,        demand\_pct, coverage\_pct, followup\_pct, recurrence\_pct, response\_pct,        service\_gap\_index, gap\_rank, confidence FROM nyc\_data.default.service\_gap\_index\_zip UNION ALL SELECT \* FROM seasonal;  |
| :---- |
